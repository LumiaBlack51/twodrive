cmake_minimum_required(VERSION 3.14)
project(desktop_instance_identity LANGUAGES CXX)
add_executable(instance_identity_test main.cpp)
target_compile_features(instance_identity_test PRIVATE cxx_std_17)
target_compile_definitions(instance_identity_test PRIVATE NOMINMAX)
if(EXISTS "${CMAKE_BINARY_DIR}/legacy_identity.h")
  add_executable(instance_identity_legacy main.cpp)
  target_compile_features(instance_identity_legacy PRIVATE cxx_std_17)
  target_compile_definitions(instance_identity_legacy PRIVATE NOMINMAX USE_LEGACY_IDENTITY)
  target_include_directories(instance_identity_legacy PRIVATE "${CMAKE_BINARY_DIR}")
endif()
